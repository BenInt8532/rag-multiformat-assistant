# Правила нарезки документов на чанки.
#
# Строка правила:   имя файла : способ, параметр=значение, параметр=значение
#
# В базу идут только файлы из папки data, у которых здесь есть строка без решётки.
# Решётка в начале строки выключает файл: в базу он не попадает.
# Уже обработанный файл заново не пересчитывается, пока не изменились его текст или правило.
#
# Способы:
#   fixed       по числу символов, не глядя на смысл (точка отсчёта)
#   paragraphs  по абзацам, длинный абзац по предложениям
#   sections    по разделам, заголовок в начале каждого чанка раздела
#   qa          вопрос и ответ - один чанк
#   lists       как paragraphs, но список не разрывается
#   table_rows  строка таблицы - один чанк с названиями столбцов
#   code        функция или класс - один чанк
#
# Параметры (необязательные):
#   size=500     размер чанка в символах
#   overlap=100  перекрытие соседних чанков в символах
#   merge=no     каждый абзац отдельным чанком (только для paragraphs; по умолчанию yes)
#   consts=yes   к чанку функции дописать настройки из начала файла, которые она использует
#                (только для code; по умолчанию no)
#
# В имени файла можно использовать звёздочку: *.py - все файлы с кодом, * - все остальные файлы.
# Если файлу подходит несколько строк, берётся первая сверху.

# Значения параметров, когда они не указаны в строке файла
default: paragraphs, size=500, overlap=100

# Единая нарезка для всех файлов в режиме CHUNKING_MODE=uniform
uniform: paragraphs, size=500, overlap=100
